cmake_minimum_required(VERSION 3.16)
project(flexcell_mtc)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(flexcell_interfaces REQUIRED)
find_package(moveit_task_constructor_core REQUIRED)
find_package(moveit_task_constructor_msgs REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(moveit_core REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tinyxml2_vendor REQUIRED)
find_package(TinyXML2 REQUIRED)
add_executable(mtc_planner src/mtc_planner.cpp)
target_compile_features(mtc_planner PUBLIC cxx_std_17)
target_link_libraries(mtc_planner tinyxml2::tinyxml2)
ament_target_dependencies(mtc_planner rclcpp flexcell_interfaces
  moveit_task_constructor_core moveit_task_constructor_msgs
  moveit_ros_planning moveit_msgs shape_msgs moveit_core geometry_msgs)
install(TARGETS mtc_planner DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(level_release_ik_test test/level_release_ik_test.cpp)
  target_compile_features(level_release_ik_test PUBLIC cxx_std_17)
  target_include_directories(level_release_ik_test PRIVATE src)
  ament_add_pytest_test(attached_point_plan_test test/test_attached_point_plan.py TIMEOUT 90
    ENV "PYTEST_ADDOPTS=-p no:cacheprovider" "PYTHONDONTWRITEBYTECODE=1")
  ament_add_pytest_test(level_release_pitch_test test/test_level_release_pitch.py TIMEOUT 120
    ENV "PYTEST_ADDOPTS=-p no:cacheprovider" "PYTHONDONTWRITEBYTECODE=1")
  ament_add_gtest(position_goal_test test/position_goal_test.cpp)
  target_include_directories(position_goal_test PRIVATE src)
  ament_target_dependencies(position_goal_test moveit_task_constructor_core moveit_core geometry_msgs)
  ament_add_gtest(planning_limits_test test/planning_limits_test.cpp)
  target_compile_features(planning_limits_test PUBLIC cxx_std_17)
  target_include_directories(planning_limits_test PRIVATE src)
  target_link_libraries(planning_limits_test tinyxml2::tinyxml2)
  ament_add_gtest(attached_object_goal_test test/attached_object_goal_test.cpp)
  target_include_directories(attached_object_goal_test PRIVATE src)
  ament_target_dependencies(attached_object_goal_test moveit_task_constructor_core moveit_core geometry_msgs)
endif()
ament_package()
