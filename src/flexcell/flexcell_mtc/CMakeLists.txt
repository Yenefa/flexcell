cmake_minimum_required(VERSION 3.16)
project(flexcell_mtc)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(flexcell_interfaces REQUIRED)
find_package(moveit_task_constructor_core REQUIRED)
find_package(moveit_task_constructor_msgs REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(moveit_core REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tinyxml2 REQUIRED)
find_package(tinyxml2_vendor REQUIRED)
find_package(rcl_interfaces REQUIRED)
add_executable(mtc_planner src/mtc_planner.cpp)
target_compile_features(mtc_planner PUBLIC cxx_std_17)
ament_target_dependencies(mtc_planner rclcpp flexcell_interfaces
  moveit_task_constructor_core moveit_task_constructor_msgs
  moveit_ros_planning moveit_msgs shape_msgs moveit_core geometry_msgs rcl_interfaces)
target_link_libraries(mtc_planner tinyxml2::tinyxml2)
install(TARGETS mtc_planner DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(position_goal_test test/position_goal_test.cpp)
  target_include_directories(position_goal_test PRIVATE src)
  ament_target_dependencies(position_goal_test moveit_task_constructor_core moveit_core geometry_msgs)
  ament_add_gtest(attached_object_goal_test test/attached_object_goal_test.cpp)
  target_include_directories(attached_object_goal_test PRIVATE src)
  ament_target_dependencies(attached_object_goal_test moveit_task_constructor_core moveit_core geometry_msgs)
  add_executable(tcp_sampler_probe test/tcp_sampler_probe.cpp)
  target_include_directories(tcp_sampler_probe PRIVATE src)
  target_compile_features(tcp_sampler_probe PUBLIC cxx_std_17)
  ament_target_dependencies(tcp_sampler_probe rclcpp moveit_ros_planning moveit_core
    moveit_task_constructor_core geometry_msgs shape_msgs flexcell_interfaces rcl_interfaces)
  target_link_libraries(tcp_sampler_probe tinyxml2::tinyxml2)
  install(TARGETS tcp_sampler_probe DESTINATION lib/${PROJECT_NAME})
  get_filename_component(MTC_SOURCE_REAL "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
  get_filename_component(FLEXCELL_SOURCE_ROOT "${MTC_SOURCE_REAL}/../../.." ABSOLUTE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME offline_tcp_probe COMMAND "${Python3_EXECUTABLE}"
    "${FLEXCELL_SOURCE_ROOT}/scripts/offline_tcp_probe.py"
    --executable "$<TARGET_FILE:tcp_sampler_probe>" --out-base "${CMAKE_CURRENT_BINARY_DIR}/offline-tcp-probes")
  set_tests_properties(offline_tcp_probe PROPERTIES TIMEOUT 100)
endif()
ament_package()
