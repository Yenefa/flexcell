cmake_minimum_required(VERSION 3.16)
project(flexcell_gazebo)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(flexcell_carry_dynamics SHARED src/CarryDynamics.cc)
add_library(flexcell_detachable_carry SHARED src/DetachableCarryGuard.cc)
add_library(flexcell_physics_trace SHARED src/PhysicsTrace.cc)
target_compile_features(flexcell_physics_trace PUBLIC cxx_std_17)
target_compile_options(flexcell_physics_trace PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(flexcell_physics_trace PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
target_compile_features(flexcell_detachable_carry PUBLIC cxx_std_17)
target_compile_options(flexcell_detachable_carry PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(flexcell_detachable_carry PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
target_compile_features(flexcell_carry_dynamics PUBLIC cxx_std_17)
target_compile_options(flexcell_carry_dynamics PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(flexcell_carry_dynamics PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS flexcell_carry_dynamics flexcell_detachable_carry flexcell_physics_trace LIBRARY DESTINATION lib)
ament_environment_hooks(hooks/plugin_path.dsv)
ament_package()
